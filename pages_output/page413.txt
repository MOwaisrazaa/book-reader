شمع شبستان رضا
يَا كَبِيرُ تَكَبَّرْتَ بِالْكِبْرِيَاءِ وَالْكِبْرِيَاءُ فِي كِبْرِيَاءِ كَبْرِيَاتِكَ يَا كَبِيرُ يَا كَرِهُمُ تَكَرَّمْتَ بِالْكَرَمِ وَلَكَرَمُ فِي كَرَمِ كَرَمِكَ يَا كَرِيمُ يَا قَدِيرُ تَعَتَرْتَ بِالْقُدْرَةِ وَالْعُدْرَةِ قَدْ رَتِكَ يَا قَدِيرُ يَا جَبَّارُ تَجَبَّرْتَ بِالْجَبَرُوتِ وَالْجَبَرُوتِ فِي جَبَرُوتِ جَبَرُوتِكَ يَا جَبَّارُ يَاتَهارُ تَرتَ بِالْقَهرِ واللهرُ فِي قَهرِ قَهْرِكَ يَاقَهَارُ مَا مَليكُ تَمَلَكْتُ بِالْمَلْكُوتِ وَالْمَلَكُوتِ فِي مَلْكُونَ مَلَكُوتِكَ يَا مَلِيكُ يَاقُدُوسُ تَعَلَّسْتَ بِالْغُدُسِ وَالْعُدُسُ فِي قُدُسِ قُدُسِكَ يَاقُدُّوسُ د دیا رو دیا يَارَبُّ تَربَّتَ بِالرَّسُونِيَّةِ وَالرَّسُونِيَّةُ فِي رَسُونِيَّةِ رَسُونِيَّتِكَ يَارَبُّ رَحِيمُ تَرَحْمَتَ بِالرَّحْمَةِ وَالرَّحْمَةُ فِي رَحْمَةِ رَحْمَتِكَ يَارَحِيمُ ریاد يَا وَهَّابُ تَوَهَّبْتَ بِالْوَهُبَةِ وَالُوهُبَةُ فِي وَهُبَةٍ وَهُبَتِكَ يَا وَهَّابُ يَامَنَّانُ تَمَتَّعُتَ بِالْمِيَّةِ وَالْمِيَّةُ فِي مِئَةِ مِئَتِكَ يَامان يَا حَكِيمُ تَحَكَمْتَ بِالْحِكْمَةِ وَالْحِكْمَةُ فِي حِكْمَةِ حِكْمَتِكَ يَا حَكِيمُ يا مُجِيدُ تَمَجَّدُتَ بِالْمَجْدِ وَالْمَجْدُ فِي مَجْدِ مَجْدِكَ يَا مَجِيدٌ يَاحَنَّانُ تَعَنَّعْتَ بِالْجِنَّةِ وَالْحِنَّهُ فِي جِنَّةِ حِيَّتِكَ يَا حَنَّانُ يَا حَمِيدٌ تَحَمَّدُت بِالْحَمْدِ وَالْحَمْدُ فِي حَمْدِ حَمْدِكَ يَا حَمِيدُ يَا حَلِهُمْ تَعَلَّمْتَ بِالْحِلْمِ وَالْحِلْمُ فِي حِلْمِ حِلْمِكَ يَا حَلِيمُ يَا قَدِيمُ تَقَدَّمْتَ بِالْعِلْمِ وَالْعِدُمُ فِي قِدْمِ قِدُمِكَ يَاقَدِيم يَا شَهِيدُ تَشَهَدتَ بِالشَّهَادَةِ وَالشَّهَادَةُ فِي شَهَادَةِ شَهَادَتِكَ يَا شَهِيدُ يَا قَرِيبُ تَقَرَّبْتَ بِالْقُرْبِ وَالْقُرْبُ فِي قُرُبِ قُرُبِكَ يَا قَرِيبُ کوردی يَا لَصِيرُ تَنصُرْتَ بِالنُّصْرَةِ وَالنُّصْرَةُ فِي مُصْرَةِ نُصُرَتِكَ يَا نَصِيرُ يَا شَكُورُ تَشْكُرْتَ بِالشَّكُرِ وَالشُّكُرُ فِي شُكُرِ شُكُرِكَ يا شُكْرِ شُكُرِكَ يا شكور يَا خَالِقُ تَخَلَعْتَ بِالْخَلْقِ وَالْخَلْقُ فِي خَلْقِ خَلْقِكَ يَا خَالِقُ
